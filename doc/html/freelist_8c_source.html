28void put_freelist_info(pmsg_level level, freelist_p fl) {
29 if (!fl) {
30 put_msg(level, "NULL free list\n");
31 return;
32 }
33 if (fl->available_total == 0) {
34 put_msg(level, "Table %s has no free space\n", schema_name(fl->sch));
35 return;
36 }
37 put_msg(level, "Table %s has free space for %d records\n",
38 schema_name(fl->sch), fl->available_total);
39 elm_p elm = fl->head;
40 while (elm) {
41 append_msg(level, " %d(%d),", elm->blk_nr, elm->available);
42 elm = elm->next;
43 }
44 append_msg(level, "\n");
45}

55void release_freelist(freelist_p fl) {
56 if (!fl)
57 return;

95int set_free_space(freelist_p fl, int blk_nr, int free_in_records) {
96 if (!fl || (blk_nr == (file_num_blocks(schema_name(fl->sch)) - 1)))
97 return 0;
98
99 elm_p elm = fl->head;
100 elm_p elm_prev = 0;
101 while (elm) {
102 if (elm->blk_nr == blk_nr) {
103 if (free_in_records == 0) {
104 fl->available_total -= elm->available;
105 delete_elm(fl, elm, elm_prev);
106 } else {
107 fl->available_total += free_in_records - elm->available;
108 elm->available = free_in_records;
109 }
110 return free_in_records;
111 }
112 if (elm->blk_nr > blk_nr) {
113 if (free_in_records > 0) {
114 fl->available_total += free_in_records;
115 insert_elm(fl, make_elm(blk_nr, free_in_records), elm_prev);
116 }
117 return free_in_records;
118 }
119 elm_prev = elm;
120 elm = elm->next;
121 }
122 fl->available_total += free_in_records;
123 insert_elm(fl, make_elm(blk_nr, free_in_records), elm_prev);
124 return free_in_records;
125}
